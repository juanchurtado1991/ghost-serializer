         reader: GhostJsonReader
     ): IntArray {
         reader.beginArray()
 
         if (reader.peekByte() == C.CLOSE_ARR) {
             reader.endArray()
             return IntArray(0)
         }
 
         val fast = tryFastIntArrayCore(

             setPosition = { reader.position = it; reader.nextTokenByte = C.RESET_TOKEN_BYTE },
         )
         if (fast != null) {
             reader.endArray()
             return fast
         }

         }

     }

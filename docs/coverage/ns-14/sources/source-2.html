     @PublishedApi
     internal fun packScanResult(
         length: Int,
         hash: Int,
         is7Bit: Boolean
     ): Long {
         var result = (length.toLong() shl SCAN_LENGTH_SHIFT) or (hash.toLong() and SCAN_HASH_MASK)
         if (is7Bit) result = result or SCAN_7BIT_BIT
         return result
     }
